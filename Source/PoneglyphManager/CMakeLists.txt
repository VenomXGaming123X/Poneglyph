# Google Fonts (SIL Open Font License) embedded into Poneglyph.exe. Downloads are OPTIONAL: if one fails the build continues
# and Poneglyph falls back to the Windows font (Segoe UI) for that choice.
set(PG_FONT_DIR "${CMAKE_CURRENT_BINARY_DIR}/fonts")
file(MAKE_DIRECTORY "${PG_FONT_DIR}")
set(PG_FONT_BASE "https://raw.githubusercontent.com/google/fonts/main/ofl")
set(PG_FONTS
  "201|inter/Inter%5Bopsz%2Cwght%5D.ttf|inter.ttf"
  "202|exo2/Exo2%5Bwght%5D.ttf|exo2.ttf"
  "203|rajdhani/Rajdhani-Regular.ttf|rajdhani.ttf"
  "204|chakrapetch/ChakraPetch-Regular.ttf|chakrapetch.ttf"
  "205|nunito/Nunito%5Bwght%5D.ttf|nunito.ttf")
set(PG_FONT_RC "")
foreach(entry ${PG_FONTS})
  string(REPLACE "|" ";" parts "${entry}")
  list(GET parts 0 fid)
  list(GET parts 1 fpath)
  list(GET parts 2 fname)
  if(NOT EXISTS "${PG_FONT_DIR}/${fname}")
    file(DOWNLOAD "${PG_FONT_BASE}/${fpath}" "${PG_FONT_DIR}/${fname}" STATUS fstatus TIMEOUT 60 TLS_VERIFY ON)
    list(GET fstatus 0 fcode)
    if(NOT fcode EQUAL 0)
      message(WARNING "Font download failed (${fname}): ${fstatus} - Poneglyph will use the Windows font for it")
      file(REMOVE "${PG_FONT_DIR}/${fname}")
    endif()
  endif()
  if(EXISTS "${PG_FONT_DIR}/${fname}")
    file(SIZE "${PG_FONT_DIR}/${fname}" fsize)
    if(fsize GREATER 20000)
      string(APPEND PG_FONT_RC "${fid} RCDATA \"${PG_FONT_DIR}/${fname}\"\n")
    else()
      file(REMOVE "${PG_FONT_DIR}/${fname}")
    endif()
  endif()
endforeach()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/pg_fonts.rc" "${PG_FONT_RC}")


# DXVK (zlib license) - the 32-bit d3d9.dll is embedded in Poneglyph.exe so the graphics-engine choice needs no extra files.
# Download is OPTIONAL: if every attempt fails the build continues and the Manager says DXVK is not part of this build.
set(PG_DXVK_DIR "${CMAKE_CURRENT_BINARY_DIR}/dxvk")
file(MAKE_DIRECTORY "${PG_DXVK_DIR}")
set(PG_DXVK_RC "")
if(NOT EXISTS "${PG_DXVK_DIR}/d3d9_x32.dll")
  foreach(ver 2.7.1 2.6.2 2.5.3 2.4.1 2.4)
    if(NOT EXISTS "${PG_DXVK_DIR}/d3d9_x32.dll")
      set(tgz "${PG_DXVK_DIR}/dxvk-${ver}.tar.gz")
      file(DOWNLOAD "https://github.com/doitsujin/dxvk/releases/download/v${ver}/dxvk-${ver}.tar.gz" "${tgz}" STATUS dst TIMEOUT 180 TLS_VERIFY ON)
      list(GET dst 0 dcode)
      if(dcode EQUAL 0)
        file(ARCHIVE_EXTRACT INPUT "${tgz}" DESTINATION "${PG_DXVK_DIR}/x" PATTERNS "dxvk-${ver}/x32/d3d9.dll")
        if(EXISTS "${PG_DXVK_DIR}/x/dxvk-${ver}/x32/d3d9.dll")
          configure_file("${PG_DXVK_DIR}/x/dxvk-${ver}/x32/d3d9.dll" "${PG_DXVK_DIR}/d3d9_x32.dll" COPYONLY)
          file(WRITE "${PG_DXVK_DIR}/version.txt" "${ver}")
          message(STATUS "DXVK ${ver} embedded")
        endif()
      else()
        message(STATUS "DXVK ${ver} download failed: ${dst}")
        file(REMOVE "${tgz}")
      endif()
    endif()
  endforeach()
endif()
if(EXISTS "${PG_DXVK_DIR}/d3d9_x32.dll")
  file(SIZE "${PG_DXVK_DIR}/d3d9_x32.dll" dsize)
  if(dsize GREATER 500000)
    set(PG_DXVK_RC "301 RCDATA \"${PG_DXVK_DIR}/d3d9_x32.dll\"\n")
  else()
    message(WARNING "DXVK file looks too small - not embedded")
  endif()
else()
  message(WARNING "DXVK could not be downloaded - this build will not offer DXVK")
endif()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/pg_dxvk.rc" "${PG_DXVK_RC}")
