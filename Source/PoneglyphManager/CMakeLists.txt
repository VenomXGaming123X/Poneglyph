# Poneglyph Manager (x64 Win32 GUI)
set(PG_ICON_DIR "${CMAKE_SOURCE_DIR}/Resources/Icons")
set(PG_RC_DEFS "")
if(EXISTS "${PG_ICON_DIR}/poneglyph_silver.ico" AND EXISTS "${PG_ICON_DIR}/poneglyph_purple.ico")
  configure_file("${PG_ICON_DIR}/poneglyph_silver.ico" "${CMAKE_CURRENT_BINARY_DIR}/poneglyph_silver.ico" COPYONLY)
  configure_file("${PG_ICON_DIR}/poneglyph_purple.ico" "${CMAKE_CURRENT_BINARY_DIR}/poneglyph_purple.ico" COPYONLY)
  set(PG_RC_DEFS "PG_HAVE_ICONS")
else()
  message(STATUS "Poneglyph icons not found in Resources/Icons - building without custom icons")
endif()
add_executable(PoneglyphManager WIN32 main.cpp PoneglyphManager.rc)
set_target_properties(PoneglyphManager PROPERTIES OUTPUT_NAME "Poneglyph")
target_include_directories(PoneglyphManager PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(PoneglyphManager PRIVATE PoneglyphCore comctl32 comdlg32 shell32 ole32 psapi uxtheme msimg32 user32 gdi32)
target_compile_definitions(PoneglyphManager PRIVATE UNICODE _UNICODE)
if(PG_RC_DEFS)
  set_source_files_properties(PoneglyphManager.rc PROPERTIES COMPILE_DEFINITIONS "${PG_RC_DEFS}")
endif()
set_source_files_properties(PoneglyphManager.rc PROPERTIES OBJECT_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/manager.manifest")
target_include_directories(PoneglyphManager PRIVATE ${CMAKE_CURRENT_BINARY_DIR})
if(MSVC)
  target_compile_options(PoneglyphManager PRIVATE /W3 /utf-8 /EHsc)
  target_link_options(PoneglyphManager PRIVATE /MANIFEST:NO)
endif()
